## Интерфейсы

- `repo_remote_url <dir>` → печатает URL remote (`origin`, иначе первый) если `<dir>` — git-репо с remote; иначе пусто. `return 0`.
- `scan_candidates <root>` → печатает по подпапке `root`: `OK\t<имя>\t<путь-репо>\t<remote-url>` или `WARN\t<имя>\t<причина>`. `return 0`.
- `cmd_scan <args...>` → разбор `--repo|--src|--dry-run|[каталог]`, запись реестра, сводка.

```bash
# ---- scan: обнаружение репозиториев ----------------------------------------
# URL remote (origin, иначе первый) если <dir> — git-репо с remote; иначе пусто.
repo_remote_url() {
  local dir="$1" url first
  [[ -d "$dir/.git" ]] || return 0
  url="$(git -C "$dir" config --get remote.origin.url 2>/dev/null || true)"
  if [[ -z "$url" ]]; then
    first="$(git -C "$dir" remote 2>/dev/null | head -1 || true)"
    [[ -n "$first" ]] && url="$(git -C "$dir" config --get "remote.$first.url" 2>/dev/null || true)"
  fi
  printf '%s' "$url"
  return 0
}

# По подпапкам <root> печатает кандидатов:
#   OK\t<имя>\t<путь-репо>\t<remote>   — репо с remote (сам или единственный в контейнере)
#   WARN\t<имя>\t<причина>             — контейнер с несколькими внутренними репо
scan_candidates() {
  local root="$1" d name url inner cnt only
  for d in "$root"/*/; do
    [[ -d "$d" ]] || continue
    d="${d%/}"
    name="$(basename "$d")"
    [[ "$name" == _* || "$name" == .* ]] && continue
    [[ -d "$d/.git" ]] || continue
    url="$(repo_remote_url "$d")"
    if [[ -n "$url" ]]; then
      printf 'OK\t%s\t%s\t%s\n' "$name" "$d" "$url"
      continue
    fi
    # контейнер (git без remote) — ищем ровно один внутренний репо с remote
    cnt=0; only=""
    for inner in "$d"/*/; do
      [[ -d "$inner" ]] || continue
      inner="${inner%/}"
      [[ -d "$inner/.git" ]] || continue
      [[ -n "$(repo_remote_url "$inner")" ]] || continue
      cnt=$((cnt+1)); only="$inner"
    done
    if [[ "$cnt" -eq 1 ]]; then
      printf 'OK\t%s\t%s\t%s\n' "$name" "$only" "$(repo_remote_url "$only")"
    elif [[ "$cnt" -gt 1 ]]; then
      printf 'WARN\t%s\t%s\n' "$name" "неоднозначный контейнер: $cnt внутренних репо с remote"
    fi
  done
  return 0
}
```

```bash
# ---- scan ------------------------------------------------------------------
# Найти репозитории в каталоге и создать записи реестра (REPO или SRC).
cmd_scan() {
  local field="" dir="" dry=0 a
  for a in "$@"; do
    a="$(strip_cr "$a")"
    case "$a" in
      --repo)    field="REPO" ;;
      --src)     field="SRC" ;;
      --dry-run) dry=1 ;;
      -*)        die "Неизвестный флаг: $a (ожидается --repo|--src|--dry-run)" ;;
      *)         dir="$a" ;;
    esac
  done
  [[ -n "$field" ]] || die "Укажи --repo или --src. Использование: pbx scan (--repo|--src) [каталог] [--dry-run]"
  if [[ -z "$dir" ]]; then
    [[ "$field" == "REPO" ]] && dir="$PROJECTS_ROOT" || dir="$WORKSPACE"
  fi
  [[ -d "$dir" ]] || die "Каталог не найден: $dir"
  mkdir -p "$PBX_REGISTRY_DIR"

  local drynote=""; [[ "$dry" -eq 1 ]] && drynote="  (dry-run)"
  c_blue "🔵 Скан $dir → поле $field$drynote"

  local created=0 skipped=0 warned=0 kind name path url f forge
  while IFS=$'\t' read -r kind name path url; do
    if [[ "$kind" == "WARN" ]]; then
      c_warn "  ⚠️  $name: $path"; warned=$((warned+1)); continue
    fi
    f="$PBX_REGISTRY_DIR/$name.conf"
    if [[ -e "$f" ]]; then
      c_warn "  • $name — пропущено (уже есть)"; skipped=$((skipped+1)); continue
    fi
    forge=""
    [[ "$url" == *github.com* ]] && forge=$'\nFORGE=github'
    if [[ "$dry" -eq 1 ]]; then
      c_green "  + $name → $field=$path (dry-run)"
    else
      printf '# Реестр pbx (scan): проект %s\n%s=%s%s\n' "$name" "$field" "$path" "$forge" > "$f"
      c_green "  + $name → $field=$path"
    fi
    created=$((created+1))
  done < <(scan_candidates "$dir")

  c_green "✅ Скан завершён: создано $created, пропущено $skipped, предупреждений $warned"
  return 0
}
```
